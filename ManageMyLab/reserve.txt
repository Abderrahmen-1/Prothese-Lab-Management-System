    if request.method == "GET":
        token = secrets.token_hex(16)
        session["csrf_token"] = token

        return render_template(
            "Log-in.html",
            csrf_token=token
        )

    if request.method == "POST":
        form_token = request.form.get("csrf_token")
        session_token = session.get("csrf_token")

        if not form_token or form_token != session_token:
            return "CSRF attack detected", 403

        new_token = secrets.token_hex(16)
        session["csrf_token"] = new_token

        email    = request.form["email"]
        username = request.form["username"]
        password = request.form["password"]

        try:
            CheckData(email, username, password)
        except InvalideNameException:
            return render_template("Log-in.html", error="Invalide Name", csrf_token= new_token)
        except InvalidePasswordException:
            return render_template("Log-in.html", error="Invalide Password", csrf_token= new_token)
        except InvalideEmailException:
            return render_template("Log-in.html", error="Invalide Email", csrf_token= new_token)

        row = ConfirmLogIn(email, username)
        if row and check_password_hash(row[2], password):
            session['id']       = row[0]
            session['username'] = username
            return render_template("HomePage.html", message=f"Welcome {session['username']}", csrf_token=form_token)
        else:
            return render_template("Log-in.html", error="Invalid Information", csrf_token= new_token)

@app.route("/DeletePatient", methods=["POST", "GET"])
def DeletePatient():
    if not is_logged_in():
        return redirect(url_for('home'))

    if request.method == "GET":
        token = secrets.token_hex(16)
        session["csrf_token"] = token

        return render_template(
            "DeletePatient.html",
            csrf_token=token
        )


    if request.method == "POST":
        form_token = request.form.get("csrf_token")
        session_token = session.get("csrf_token")

        if not form_token or form_token != session_token:
            return "CSRF attack detected", 403

        new_token = secrets.token_hex(16)
        session["csrf_token"] = new_token


        PatientF_deleted = request.form["DeleteF"]
        Confirm_PatientF = request.form["ConfirmDeletePatientF"]
        PatientL_deleted = request.form["DeleteL"]
        Confirm_PatientL = request.form["ConfirmDeletePatientL"]
        Patient_unique   = request.form["uniquepatient"]
        Confirm_Unique   = request.form["confirmunique"]

        if not checkIntValue(Patient_unique) or not checkIntValue(Confirm_Unique):
            return render_template("DeletePatient.html", message="Invalid Unique number", csrf_token=new_token)

        if not Patient_unique.isdigit() or len(Patient_unique) != 10:
            return render_template("DeletePatient.html", message="Invalid Unique number", csrf_token=new_token)
        if (PatientF_deleted != Confirm_PatientF or
                PatientL_deleted != Confirm_PatientL or
                Patient_unique != Confirm_Unique):
            return render_template("DeletePatient.html", message="Invalid Confirmation", csrf_token=new_token)

        conn = get_connection()
        cursor = conn.cursor(buffered=True)
        try:
            cursor.execute("""
                    SELECT versa , b9aw , MONTH(la_date) , YEAR(la_date)  FROM patient
                    WHERE patient_firstname = %s AND patient_lastname = %s
                    AND patient_number = %s AND user_id = %s
                    """,(PatientF_deleted, PatientL_deleted, Patient_unique, session['id']))

            result = cursor.fetchone()

            versa = result[0]
            b9aw = result[1]
            month = result[2]
            year = result[3]

            cursor.execute("""
                                UPDATE patient_stat
                                SET
                                khalsin = khalsin - %s ,
                                machi_khalsin = machi_khalsin - %s
                                WHERE user_id = %s AND month = %s AND year = %s
                                """, (versa, b9aw, session['id'], month, year))


            cursor.execute("DELETE FROM khedma WHERE user_id = %s AND khedma_patient = %s",
                           (session['id'], Patient_unique))

            cursor.execute("""
                DELETE FROM patient
                WHERE patient_firstname = %s AND patient_lastname = %s
                  AND patient_number = %s AND user_id = %s
            """, (PatientF_deleted, PatientL_deleted, Patient_unique, session['id']))

            cursor.execute("""
                            UPDATE patient_history
                            SET status = 'deleted'
                            WHERE patient_firstname = %s AND patient_lastname = %s
                            AND patient_number = %s AND user_id = %s
            """, (PatientF_deleted, PatientL_deleted, Patient_unique, session['id']))

            conn.commit()

            cursor.execute("""
                            SELECT * FROM patient
                            WHERE user_id = %s """, (session['id'],))
            data = cursor.fetchall()

            return render_template("PatientsFile.html", message="Patient Deleted Successfully", data=data, csrf_token=new_token)

        except Exception as e :
            conn.rollback()
            return render_template("DeletePatient.html", message=f"error {str(e)}", csrf_token=new_token)

        finally:
            cursor.close()
            conn.close()


@app.route("/ModifyVersaPatient", methods=["POST" , "GET"])
def ModifyVersaPatient():
    if not is_logged_in():
        return redirect(url_for('home'))

    if request.method == "GET":
        token = secrets.token_hex(16)
        session["csrf_token"] = token

        return render_template(
            "ModifyPatientVersa.html",
            csrf_token=token
        )


    if request.method == "POST":
        form_token = request.form.get("csrf_token")
        session_token = session.get("csrf_token")

        if not form_token or form_token != session_token:
            return "CSRF attack detected", 403

        new_token = secrets.token_hex(16)
        session["csrf_token"] = new_token


        Patient_Id = request.form['PatientId']
        VersaValue = request.form['VersaValue']
        ConfirmVersa = request.form['ConfirmVersaValue']

        if not Patient_Id or not Patient_Id.isdigit() or int(Patient_Id) < 0 or len(Patient_Id) < 10 :
            return render_template("ModifyPatientVersa.html",
                                   message="Invalide ID ",
                                   csrf_token=new_token)


        if not checkValue(VersaValue) or not checkValue(ConfirmVersa):
            return render_template("ModifyPatientVersa.html", message="Invalide versement values", csrf_token=new_token)

        if not VersaValue :
            VersaValue = 0
        if not ConfirmVersa :
            ConfirmVersa = 0

        VersaValue = float(VersaValue)
        ConfirmVersa = float(ConfirmVersa)

        if VersaValue != ConfirmVersa :
            return render_template("ModifyPatientVersa.html" , message="The Versa and its Confirmation must be the same value ", csrf_token=new_token)

        conn = get_connection()
        cursor = conn.cursor(buffered=True)

        try :
            cursor.execute("""
                            SELECT  versa , b9aw , khalsa , MONTH(la_date) , YEAR(la_date)  FROM patient
                            WHERE user_id = %s AND patient_number = %s """
                           ,(session['id'], Patient_Id ))

            result = cursor.fetchone()
            if not result:
                return render_template("ModifyPatientVersa.html" , message="Patient NOt Found", csrf_token=new_token)

            total = float(result[0]) + float(result[1])
            oldVersa = float(result[0])
            oldB9aw = float(result[1])
            oldKhalsa = result[2]
            month = result[3]
            year = result[4]

            if oldKhalsa.lower() == "yes" :
                return render_template("ModifyPatientVersa.html" ,
                                       message="The Patient already did payed his khedma ", csrf_token=new_token)

            new_versa = oldVersa + VersaValue

            if new_versa == total :
                b9aw = 0

                cursor.execute("""
                                UPDATE patient
                                SET
                                khalsa = "yes" ,
                                b9aw = %s ,
                                versa = %s
                                WHERE user_id = %s AND patient_number = %s
                                """,(b9aw , new_versa , session['id'] , Patient_Id))

                cursor.execute("""
                                UPDATE patient_history
                                SET
                                versement = %s , b9aw = %s
                                WHERE user_id = %s AND patient_number = %s """,
                               (new_versa, b9aw, session['id'], Patient_Id))

                cursor.execute("""
                               UPDATE patient_stat
                               SET
                                khalsin = khalsin + %s ,
                                machi_khalsin = machi_khalsin - %s
                               WHERE
                                user_id = %s AND month = %s AND year = %s """
                               ,(VersaValue, oldB9aw, session['id'], month, year ))

                cursor.execute("""
                                UPDATE khedma
                                SET khedma_khalsa = 'yes'
                                WHERE user_id = %s AND khedma_patient = %s  """,(session['id'], Patient_Id ))

                conn.commit()

            elif new_versa > total :
                return render_template("ModifyPatientVersa.html" , message="Invalide Versa Amount", csrf_token=new_token)

            else :
                b9aw = total - new_versa

                cursor.execute("""
                                 UPDATE patient
                                 SET
                                 b9aw = %s ,
                                 versa = %s
                                 WHERE user_id = %s AND patient_number = %s """,
                               (b9aw, new_versa , session['id'], Patient_Id))

                cursor.execute("""
                                UPDATE patient_history
                                SET
                                versement = %s , b9aw = %s
                                WHERE user_id = %s AND patient_number = %s """,
                               (new_versa, b9aw, session['id'], Patient_Id))

                cursor.execute("""
                                 UPDATE patient_stat
                                 SET
                                 khalsin = khalsin + %s ,
                                 machi_khalsin = machi_khalsin - %s
                                 WHERE
                                 user_id = %s AND month = %s AND year = %s """
                               , (VersaValue, VersaValue, session['id'], month, year) )

                cursor.execute("""
                                DELETE FROM patient
                                WHERE finished = %s AND khalsa = %s AND user_id = %s
                                """,('yes', 'yes', session['id']))

                cursor.execute("""
                                SELECT * FROM patient
                                WHERE user_id = %s """,(session['id'],))
                result = cursor.fetchall()

                conn.commit()

        except Exception as e :
            conn.rollback()
            return render_template("ModifyPatientVersa.html", message=f"error {str(e)}", csrf_token=new_token)

        finally :
            cursor.close()
            conn.close()

        return render_template("PatientsFile.html", message="Informations Updated", data=result)

@app.route("/UpdatePatientStatus" , methods=["POST" , "GET"])
def UpdatePatientStatus():
    if not is_logged_in():
        return jsonify({
            'success' :False,
            'message' : "user not allowed"
        }),401


    data = request.get_json()

    if not data :
        return jsonify({
            'success' : False,
            'message' : 'data not recieved'
        }),401

    patientID = data.get('patient_number')
    status = data.get('status')

    if not patientID or not status :
        return jsonify({
            'success' : False,
            'message' : 'Patient number and status are required'
        }),403

    allowed_status = [
        'in process',
        'prothesist',
        'montage',
        'esseyage',
        'finished'
    ]

    if status not in allowed_status :
        return jsonify({
            'success' : False,
            'message' : 'Invalide status value'
        }),403

    conn = get_connection()
    cursor = conn.cursor(buffered=True)

    try :

        cursor.execute("""
                        SELECT 1
                        FROM patient
                        WHERE user_id = %s AND patient_number = %s """,
                       (session['id'], patientID))

        found = cursor.fetchone()

        if not found :
            return jsonify({
                'success' : False,
                'message' : 'Patient not found'
            }),403

        cursor.execute("""
                        UPDATE patient
                        SET status = %s
                        WHERE user_id = %s AND patient_number = %s """,
                       (status, session['id'], patientID))
        conn.commit()

        print("Patient ID:", patientID)
        print("New status:", status)
        print("Rows updated:", cursor.rowcount)

    except Exception as e:
        conn.rollback()
        return jsonify({
            'success' : False,
            'message' : f'error {str(e)}'
        })

    finally :
        cursor.close()
        conn.close()

    return jsonify({
        'success' : True,
        'message' : 'Status updated successfully'
    })

    if not is_logged_in():
        return jsonify({
            'success' : False,
            "error": "Unauthorized"
        }), 401

    data = request.get_json()
    new_versement = data.get('versement')

    try:
        new_versement = float(new_versement)
    except (ValueError, TypeError):
        return jsonify({
            'success': False,
            'error': 'The versement must be a valid number'
        }), 400

    if new_versement < 0:
        return jsonify({
            'success': False,
            'error': 'The versement must be a positive value'
        }), 400

    conn = get_connection()
    cursor = conn.cursor(buffered=True)

    try :

        cursor.execute("""
                        SELECT sel3a_price, versement, b9aw, khalsa
                        FROM sel3a
                        WHERE user_id = %s AND id = %s """,(session['id'], sel3a_id))
        result = cursor.fetchone()

        final = float(new_versement) + float(result[1])

        if final == float(result[0]) :
            cursor.execute("""
                            UPDATE sel3a
                            SET versement = %s , b9aw = %s , khalsa = 'yes'
                            WHERE user_id = %s AND id = %s """,
                           (result[0], 0, session['id'], sel3a_id))
            conn.commit()

            return jsonify({
                'success': True,
                'message' : 'Data Updated'
            }), 200

        elif final > float(result[0]) :

            return jsonify({
                'success': False,
                'error': 'Invalide versement value'
            }), 401

        else :
            b9aw = float(result[0]) - final
            cursor.execute("""
                            UPDATE sel3a
                            SET versement = %s , b9aw = %s
                            WHERE user_id = %s AND id = %s """,
                           (final, b9aw, session['id'], sel3a_id))

            return jsonify({
                'success': True,
                'message': 'Data Updated'
            }), 200

    except Exception as e :
        conn.rollback()

        }),500

    finally :

@app.route("/AllTotalAccount")
def AllTotal():
    if not is_logged_in():
        return redirect(url_for('home'))

    conn = get_connection()
    cursor = conn.cursor(buffered=True)

    try :

        cursor.execute("""
        SELECT
            month,
            year,

            SUM(CASE WHEN source='patients' THEN khalsin ELSE 0 END) AS patient_khalsin,
            SUM(CASE WHEN source='patients' THEN b9aw ELSE 0 END) AS patient_b9aw,

            SUM(CASE WHEN source='dentists' THEN khalsin ELSE 0 END) AS dentist_khalsin,
            SUM(CASE WHEN source='dentists' THEN b9aw ELSE 0 END) AS dentist_b9aw,

            SUM(CASE WHEN source='prothesists' THEN khalsin ELSE 0 END) AS pro_khalsin,
            SUM(CASE WHEN source='prothesists' THEN b9aw ELSE 0 END) AS pro_b9aw,

            SUM(CASE WHEN source='sel3a' THEN khalsin ELSE 0 END) AS sel3a_khalsa,
            SUM(CASE WHEN source='sel3a' THEN b9aw ELSE 0 END) AS sel3a_b9aw,

            SUM(CASE WHEN source='others' THEN khalsin ELSE 0 END) AS other_khalsin,
            SUM(CASE WHEN source='others' THEN b9aw ELSE 0 END) AS other_b9aw,

            SUM(CASE
                    WHEN source='patients' THEN khalsin - b9aw
                    WHEN source='dentists' THEN khalsin - b9aw
                    WHEN source='prothesists' THEN -(khalsin + b9aw)
                    WHEN source='sel3a' THEN -(khalsin + b9aw)
                    WHEN source='others' THEN -(khalsin + b9aw)
                    ELSE 0 END) AS Net_Profets
        FROM (

            SELECT month, year,
                   SUM(khalsin) AS khalsin,
                   SUM(machi_khalsin) AS b9aw,
                   'patients' AS source
            FROM patient_stat
            WHERE user_id = %s
            GROUP BY month, year

            UNION ALL

            SELECT month, year,
                   SUM(khalsin),
                   SUM(machi_khalsin),
                   'dentists'
            FROM dentist_stat
            WHERE user_id = %s
            GROUP BY month, year

            UNION ALL

            SELECT month, year,
                   SUM(khalsin),
                   SUM(machi_khalsin),
                   'prothesists'
            FROM prothesist_stat
            WHERE user_id = %s
            GROUP BY month, year

            UNION ALL

            SELECT
                MONTH(buy_date),
                YEAR(buy_date),
                SUM(CASE WHEN khalsa='yes' THEN versement ELSE 0 END) AS khalsa,
                SUM(CASE WHEN khalsa='no' THEN b9aw ELSE 0 END) AS b9aw,
                'sel3a'
            FROM sel3a
            WHERE user_id = %s
            GROUP BY MONTH(buy_date), YEAR(buy_date)

            UNION ALL

            SELECT
                 month , year ,
                 SUM(khalsin) AS khalsin ,
                 SUM(machi_khalsin) AS b9aw ,
                 'others' AS source
                 FROM other_services_stats
                 WHERE user_id = %s
                 GROUP BY month , year

        ) AS combined

        GROUP BY month, year
        ORDER BY year, month
        """, (session['id'], session['id'], session['id'], session['id'], session['id']))

        result = cursor.fetchall()
        return render_template("AllTotalStats.html", result=result)

    finally :
        cursor.close()
        conn.close()

@app.route("/AddService", methods=["GET", "POST"])
def AddService():
    if not is_logged_in():
        return redirect(url_for('home'))

    if request.method == "GET":
        token = secrets.token_hex(16)
        session["csrf_token"] = token

        return render_template(
            "AddService.html",
            csrf_token=token
        )

    if request.method == "POST":
        form_token = request.form.get("csrf_token")
        session_token = session.get("csrf_token")

        if not form_token or form_token != session_token:
            return "CSRF attack detected", 403

        new_token = secrets.token_hex(16)
        session["csrf_token"] = new_token


        name = request.form.get('ServiceName', '').strip()
        number = request.form.get('ServiceNumber', '').strip()
        function = request.form.get('ServiceFunction', '').strip()
        date = request.form.get('ServiceDate')
        khalsa = request.form.get('ServiceKhalsa', 'no').lower()
        price = float(request.form.get('ServicePrice', 0))

        # ---------------- VALIDATION ----------------
        if not name:
            return render_template("AddService.html", message="Invalid Name", csrf_token=new_token)

        if not number.isdigit() or len(number) != 10:
            return render_template("AddService.html", message="Invalid Number", csrf_token=new_token)

        if not function:
            return render_template("AddService.html", message="Invalid Function", csrf_token=new_token)

        if khalsa not in ['yes', 'no']:
            return render_template("AddService.html", message="Khalsa must be yes or no", csrf_token=new_token)

        if price < 0:
            return render_template("AddService.html", message="Invalid Price", csrf_token=new_token)

        if khalsa == 'no':
            versa = request.form.get('ServiceVersa', 'no').lower()

            if versa not in ['yes', 'no']:
                return render_template("AddService.html", message="Versa must be yes or no", csrf_token=new_token)

            if versa == 'yes':
                Vprice = float(request.form.get('VersaService', '0'))

                if Vprice < 0:
                    return render_template("AddService.html", message="Invalid Price", csrf_token=new_token)

                if Vprice > price:
                    return render_template("AddService.html", message="Versa > Price not allowed", csrf_token=new_token)

                if Vprice == price :
                    b9aw = 0
                    khalsa = "yes"

                b9aw = price - Vprice
            else:
                Vprice = 0
                b9aw = price
        else:
            Vprice = price
            b9aw = 0

        # ---------------- DATABASE ----------------
        conn = get_connection()
        cursor = conn.cursor(buffered=True)

        try:
            cursor.execute("""
                INSERT INTO others_services
                (name, number, date, price, khalsa, versement, b9aw, user_id, k_function)
                VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s)
            """, (name, number, date, price, khalsa, Vprice, b9aw, session['id'], function))

            conn.commit()

            d = datetime.strptime(date, "%Y-%m-%d")
            month = d.month
            year = d.year

            cursor.execute("""
                SELECT id FROM other_services_stats
                WHERE user_id = %s AND month = %s AND year = %s AND number = %s
            """, (session['id'], month, year, number))

            result = cursor.fetchone()

            if result:
                cursor.execute("""
                    UPDATE other_services_stats
                    SET
                        machi_khalsin = machi_khalsin + %s,
                        khalsin = khalsin + %s,
                        total_khedma = total_khedma + 1
                    WHERE user_id = %s AND number = %s AND month = %s AND year = %s
                """, (b9aw, Vprice, session['id'], number, month, year))
            else:
                cursor.execute("""
                    INSERT INTO other_services_stats
                    (Name, Number, total_khedma, khalsin, machi_khalsin, month, year, user_id)
                    VALUES (%s, %s, %s, %s, %s, %s, %s, %s)
                """, (name, number, 1, Vprice, b9aw, month, year, session['id']))

            conn.commit()

            return render_template("OtherServices.html", message="Service Added Successfully", csrf_token=new_token)

        finally:
            cursor.close()
            conn.close()

@app.route("/webhook_chargily" , methods=['POST'])
def webhook_chargily():
    data = request.json

    # Vérifier que c'est bien Chargily qui envoie (sécurité)
    signature = request.headers.get("signature", "")
    payload = request.get_data()
    secret = os.getenv("CHARGILY_WEBHOOK_SECRET").encode()
    expected = hmac.new(secret, payload, hashlib.sha256).hexdigest()

    if not hmac.compare_digest(signature, expected):
        return "Invalide", 403  # ← quelqu'un essaie de tricher

    # Si le paiement est confirmé
    if data.get("status") == "paid":
        metadate = data.get("metadata" , {})
        user_id = metadate.get("user_id")
        if not user_id :
            return "Missing user ",400

        conn = get_connection()
        cursor = conn.cursor(buffered=True)
        try:
            cursor.execute(""" SELECT date_inscription FROM users WHERE id = %s """,(session['id'],))
            row = cursor.fetchone

            if row and row[0] and row[0] > datetime.now() :
                nouvelle_date = row[0] + timedelta(days=30)
            else :
                nouvelle_date = datetime.now() + timedelta(days=30)

            cursor.execute("""
                    UPDATE users
                    SET date_inscription = %s
                    WHERE id = %s
                """, (nouvelle_date, user_id))
            conn.commit()
        finally:
            cursor.close()
            conn.close()

    return "OK", 200
